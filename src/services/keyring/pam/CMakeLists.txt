find_package(PkgConfig REQUIRED)
pkg_check_modules(PAM REQUIRED IMPORTED_TARGET GLOBAL pam)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET GLOBAL libsystemd)
pkg_get_variable(KEYRING_SYSTEM_UNIT_DIR systemd systemdsystemunitdir)
set(KEYRING_HELPER_PATH "${KDE_INSTALL_FULL_LIBEXECDIR}/qindaqt-keyring-pam-helper")
set(KEYRING_LAUNCHER_PATH "${KDE_INSTALL_FULL_LIBEXECDIR}/qindaqt-keyring-launch")
set(KEYRING_SYSTEM_UNIT "${KEYRING_SYSTEM_UNIT_DIR}/qindaqt-keyring@.service")
function(keyring_pam_target target)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(${target} PRIVATE QindaQt::KeyringStorage)
    target_compile_definitions(${target} PRIVATE QINDAQT_PAM_HELPER="${KEYRING_HELPER_PATH}"
        QINDAQT_KEYRING_LAUNCHER="${KEYRING_LAUNCHER_PATH}" QINDAQT_KEYRING_SYSTEM_UNIT="${KEYRING_SYSTEM_UNIT}"
        QINDAQT_KEYRING_DAEMON="${CMAKE_INSTALL_FULL_BINDIR}/qindaqt-keyring")
    qindaqt_enable_warnings(${target})
endfunction()
add_library(pam_qindaqt_keyring MODULE module.cpp bridge_process.cpp transport.cpp system_owner.cpp)
set_target_properties(pam_qindaqt_keyring PROPERTIES PREFIX "")
keyring_pam_target(pam_qindaqt_keyring)
target_link_libraries(pam_qindaqt_keyring PRIVATE PkgConfig::PAM PkgConfig::SYSTEMD)
add_executable(qindaqt-keyring-pam-helper helper.cpp transport.cpp system_owner.cpp)
keyring_pam_target(qindaqt-keyring-pam-helper)
target_link_libraries(qindaqt-keyring-pam-helper PRIVATE PkgConfig::SYSTEMD)
add_executable(qindaqt-keyring-launch launcher.cpp transport.cpp)
keyring_pam_target(qindaqt-keyring-launch)
configure_file(qindaqt-keyring@.service.in qindaqt-keyring@.service @ONLY)
install(TARGETS pam_qindaqt_keyring LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/security" COMPONENT QindaQtKeyringPam)
install(TARGETS qindaqt-keyring-pam-helper qindaqt-keyring-launch RUNTIME DESTINATION "${KDE_INSTALL_LIBEXECDIR}" COMPONENT QindaQtKeyringPam)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/qindaqt-keyring@.service" DESTINATION "${KEYRING_SYSTEM_UNIT_DIR}" COMPONENT QindaQtKeyringPam)
