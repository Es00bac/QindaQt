qt_add_library(qindaqt_lock_authentication STATIC authentication.cpp authentication.h attempt_coordinator.cpp attempt_coordinator.h)
target_include_directories(qindaqt_lock_authentication PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(qindaqt_lock_authentication PUBLIC cxx_std_20)
set_target_properties(qindaqt_lock_authentication PROPERTIES CXX_EXTENSIONS OFF)
qindaqt_enable_warnings(qindaqt_lock_authentication)
if(QINDAQT_BUILD_PRODUCTION_SHELL OR BUILD_TESTING)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LockPAM REQUIRED IMPORTED_TARGET pam)
endif()
if(QINDAQT_BUILD_PRODUCTION_SHELL)
    qt_add_library(qindaqt_lock_native_pam STATIC native_pam_transaction.cpp native_pam_transaction.h)
    target_link_libraries(qindaqt_lock_native_pam PUBLIC qindaqt_lock_authentication PRIVATE PkgConfig::LockPAM)
    qindaqt_enable_warnings(qindaqt_lock_native_pam)
endif()
if(BUILD_TESTING)
    qt_add_executable(tst_lock_authentication tests/tst_authentication.cpp)
    target_link_libraries(tst_lock_authentication PRIVATE qindaqt_lock_authentication Qt6::Test)
    add_test(NAME lock.authentication COMMAND tst_lock_authentication)
endif()

if(BUILD_TESTING)
    add_library(qindaqt-private-pam-witness MODULE tests/private_pam_module.cpp)
    target_link_libraries(qindaqt-private-pam-witness PRIVATE PkgConfig::LockPAM)
    qt_add_executable(tst_native_pam_conversation tests/tst_native_pam_conversation.cpp native_pam_transaction.cpp)
    target_link_libraries(tst_native_pam_conversation PRIVATE qindaqt_lock_authentication Qt6::Test PkgConfig::LockPAM)
    target_compile_definitions(tst_native_pam_conversation PRIVATE QINDAQT_PRIVATE_PAM_FIXTURE
        QINDAQT_PRIVATE_PAM_MODULE_PATH="$<TARGET_FILE:qindaqt-private-pam-witness>")
    add_dependencies(tst_native_pam_conversation qindaqt-private-pam-witness)
    qindaqt_enable_warnings(tst_native_pam_conversation)
    add_test(NAME lock.native-pam-conversation COMMAND tst_native_pam_conversation)
endif()
