cmake_minimum_required(VERSION 3.25)
project(QindaQtAndroidApplicationsComponent LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(android_root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${android_root}/cmake")
include(QindaQtCompiler)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui DBus Qml Quick QuickControls2)
if(BUILD_TESTING)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Test)
endif()
qt_standard_project_setup(REQUIRES 6.11)
set(QINDAQT_LAUNCHER_PURE_ONLY ON)
set(QINDAQT_BUILD_ANDROID_NATIVE_COLLECTOR ON)
foreach(path IN ITEMS shell/launcher application_catalog services/android_runtime apps/android_manager)
    add_subdirectory("${android_root}/src/${path}" "${CMAKE_BINARY_DIR}/src/${path}")
endforeach()
if(BUILD_TESTING)
    add_subdirectory("${android_root}/tests/services/android_runtime/bootstrap" bootstrap-tests)
    add_subdirectory("${android_root}/tests/apps/android_manager" manager-tests)
endif()
