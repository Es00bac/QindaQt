if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.25)
    project(QindaQtWindowGroupingTests LANGUAGES CXX)
    include(CTest)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Core Test)
    set(QINDAQT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
    function(qindaqt_enable_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endfunction()
    add_subdirectory("${QINDAQT_ROOT}/src/core" "${CMAKE_BINARY_DIR}/core")
    add_subdirectory("${QINDAQT_ROOT}/src/hybrid" "${CMAKE_BINARY_DIR}/hybrid")
    add_subdirectory("${QINDAQT_ROOT}/src/window_management/grouping" "${CMAKE_BINARY_DIR}/grouping")
    add_subdirectory("${QINDAQT_ROOT}/tests/hybrid" "${CMAKE_BINARY_DIR}/hybrid-tests")
endif()
foreach(behavior planner activation)
    qt_add_executable(window_grouping_${behavior} tst_${behavior}.cpp)
    target_link_libraries(window_grouping_${behavior} PRIVATE QindaQt::WindowManagementGrouping Qt6::Test)
    target_compile_features(window_grouping_${behavior} PRIVATE cxx_std_20)
    set_target_properties(window_grouping_${behavior} PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
    qindaqt_enable_warnings(window_grouping_${behavior})
    add_test(NAME qindaqt.window-grouping-${behavior} COMMAND window_grouping_${behavior})
    set_tests_properties(qindaqt.window-grouping-${behavior} PROPERTIES ENVIRONMENT "QT_FATAL_WARNINGS=1")
endforeach()
