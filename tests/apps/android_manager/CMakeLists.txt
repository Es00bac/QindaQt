if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.25)
    project(QindaQtAndroidManagerTests LANGUAGES CXX)
    include(CTest)
    include(GNUInstallDirs)
    list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
    include(QindaQtCompiler)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test)
    qt_standard_project_setup(REQUIRES 6.11)
    set(QINDAQT_LAUNCHER_PURE_ONLY ON)
    add_subdirectory("../../../src/shell/launcher" "${CMAKE_BINARY_DIR}/src/shell/launcher")
    add_subdirectory("../../../src/application_catalog" "${CMAKE_BINARY_DIR}/src/application_catalog")
    add_subdirectory("../../../src/services/android_runtime" "${CMAKE_BINARY_DIR}/src/services/android_runtime")
    add_subdirectory("../../../src/apps/android_manager" "${CMAKE_BINARY_DIR}/src/apps/android_manager")
endif()
get_filename_component(android_manager_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/isolated/runtime" "${CMAKE_CURRENT_BINARY_DIR}/isolated/home" "${CMAKE_CURRENT_BINARY_DIR}/isolated/config" "${CMAKE_CURRENT_BINARY_DIR}/isolated/data" "${CMAKE_CURRENT_BINARY_DIR}/artifacts")
file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/isolated/runtime" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)
set(android_manager_environment
    "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=software;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Basic;QT_QPA_PLATFORMTHEME=;QT_FATAL_WARNINGS=1;HOME=${CMAKE_CURRENT_BINARY_DIR}/isolated/home;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/isolated/config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/isolated/data;XDG_RUNTIME_DIR=${CMAKE_CURRENT_BINARY_DIR}/isolated/runtime;DBUS_SESSION_BUS_ADDRESS=unix:path=/nonexistent;DBUS_SYSTEM_BUS_ADDRESS=unix:path=/nonexistent;DISPLAY=;WAYLAND_DISPLAY=")
qt_add_executable(qindaqt_android_manager_controller_tests tst_application_controller.cpp)
target_link_libraries(qindaqt_android_manager_controller_tests PRIVATE QindaQt::AndroidManager Qt6::Test)
qindaqt_enable_warnings(qindaqt_android_manager_controller_tests)
add_test(NAME apps.android-manager.controller COMMAND qindaqt_android_manager_controller_tests)
set_tests_properties(apps.android-manager.controller PROPERTIES TIMEOUT 10 ENVIRONMENT "${android_manager_environment}")
if(NOT TARGET qindaqt-android-applications)
    return()
endif()
qt_add_executable(qindaqt_android_manager_ui_tests tst_ui.cpp)
target_link_libraries(qindaqt_android_manager_ui_tests PRIVATE QindaQt::AndroidManager Qt6::Test Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(qindaqt_android_manager_ui_tests PRIVATE
    ANDROID_MANAGER_MAIN_QML="${android_manager_root}/src/apps/android_manager/qml/Main.qml"
    ANDROID_MANAGER_ARTIFACTS="${CMAKE_CURRENT_BINARY_DIR}/artifacts")
qindaqt_enable_warnings(qindaqt_android_manager_ui_tests)
add_test(NAME apps.android-manager.ui COMMAND qindaqt_android_manager_ui_tests)
set_tests_properties(apps.android-manager.ui PROPERTIES TIMEOUT 10 ENVIRONMENT "${android_manager_environment}")
add_test(NAME apps.android-manager.compiled-smoke COMMAND qindaqt-android-applications --smoke)
set_tests_properties(apps.android-manager.compiled-smoke PROPERTIES TIMEOUT 10 ENVIRONMENT "${android_manager_environment}")
