cmake_minimum_required(VERSION 3.25)
project(QindaQtFileRecoveryHarness LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QINDAQT_ENABLE_STRICT_WARNINGS "Treat warnings as errors" ON)
get_filename_component(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(owner_root "${source_root}/src/apps/file_manager")
set(test_root "${source_root}/tests/apps/file_manager")
include("${source_root}/cmake/QindaQtCompiler.cmake")
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
qt_standard_project_setup()
# Consume the exact existing desktop boundary source registration, rather than
# copying its implementation or building AppShell/KIO/desktop presentation.
file(READ "${owner_root}/CMakeLists.txt" owner_registry)
string(REGEX MATCH "target_sources\\([\n\r ]*qindaqt_file_manager_desktop_boundary[\n\r ]*PRIVATE([^)]*)\\)"
    owner_match "${owner_registry}")
if(NOT owner_match)
    message(FATAL_ERROR "Owning desktop boundary registration was not found")
endif()
set(owner_sources "${CMAKE_MATCH_1}")
string(REGEX REPLACE "[\n\r\t ]+PUBLIC[\n\r\t ]+" " " owner_sources "${owner_sources}")
separate_arguments(owner_sources UNIX_COMMAND "${owner_sources}")
list(TRANSFORM owner_sources PREPEND "${owner_root}/")
qt_add_library(qindaqt_file_manager_desktop_boundary STATIC ${owner_sources})
target_include_directories(qindaqt_file_manager_desktop_boundary PUBLIC "${owner_root}")
target_link_libraries(qindaqt_file_manager_desktop_boundary PUBLIC Qt6::Core Qt6::Gui)
qindaqt_enable_warnings(qindaqt_file_manager_desktop_boundary)
foreach(behavior record mount manifest cross_volume replacement)
    qt_add_executable(ed05_${behavior}_tests "${test_root}/tst_recovery_${behavior}.cpp")
    target_include_directories(ed05_${behavior}_tests PRIVATE "${test_root}")
    target_link_libraries(ed05_${behavior}_tests PRIVATE qindaqt_file_manager_desktop_boundary Qt6::Test)
    qindaqt_enable_warnings(ed05_${behavior}_tests)
    if(behavior STREQUAL "cross_volume")
        add_test(NAME recovery-${behavior} COMMAND "${Python3_EXECUTABLE}"
            "${test_root}/run_recovery_owned.py" $<TARGET_FILE:ed05_${behavior}_tests>)
    else()
        add_test(NAME recovery-${behavior} COMMAND ed05_${behavior}_tests)
    endif()
endforeach()
foreach(behavior local_mutation_backend home_trash mutation_controller mutation_output copy_safety)
    qt_add_executable(ed05_${behavior}_tests "${test_root}/tst_${behavior}.cpp")
    target_include_directories(ed05_${behavior}_tests PRIVATE "${test_root}")
    target_link_libraries(ed05_${behavior}_tests PRIVATE qindaqt_file_manager_desktop_boundary Qt6::Test)
    qindaqt_enable_warnings(ed05_${behavior}_tests)
    add_test(NAME regression-${behavior} COMMAND ed05_${behavior}_tests)
endforeach()
add_library(ed05_io_fault SHARED "${test_root}/recovery_io_fault_shim.cpp")
target_link_libraries(ed05_io_fault PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(ed05_io_fault PROPERTIES PREFIX "")
qindaqt_enable_warnings(ed05_io_fault)
target_link_libraries(ed05_cross_volume_tests PRIVATE ${CMAKE_DL_LIBS})
get_property(registered DIRECTORY PROPERTY TESTS)
set_tests_properties(${registered} PROPERTIES TIMEOUT 60
    ENVIRONMENT_MODIFICATION "DISPLAY=unset:;WAYLAND_DISPLAY=unset:"
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic;DBUS_SESSION_BUS_ADDRESS=unix:path=/nonexistent;DBUS_SYSTEM_BUS_ADDRESS=unix:path=/nonexistent")
set_property(TEST recovery-cross_volume APPEND PROPERTY ENVIRONMENT "LD_PRELOAD=$<TARGET_FILE:ed05_io_fault>")

add_test(NAME recovery-runner-injected COMMAND "${Python3_EXECUTABLE}" "${test_root}/test_recovery_owned.py")
set_tests_properties(recovery-runner-injected PROPERTIES TIMEOUT 10 LABELS "fake")
